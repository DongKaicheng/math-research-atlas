\begin{lemma}[Smooth scalar integration by parts; F8MG-S]
\label{lem:smooth}
For ordinary smooth \(h\) on a neighbourhood of the compact support
of \(\eta\in\A^{n-1,n-1}_{U,c}\),
\begin{equation}\label{eq:smooth}
                    \int_U h\,\dd\eta=\int_U\dd h\wedge\eta.
\end{equation}
The compact support in \(U\subseteq\Int(X/K)\) is retained.
\end{lemma}
