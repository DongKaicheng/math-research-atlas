\begin{corollary}[Two scalar conventions for finite intersection algebras]
\label{f9ds:zerodim}
If $C$ is any finite-dimensional $K$-algebra, then
\[
 \dim_L(C\otimes_KL)=\dim_K C,\qquad
 \dim_K(C\otimes_KL)=n\dim_K C.
\]
For any finite-dimensional algebra $D$ over a field $H$,
\[
 \dim_H D
   =\sum_j\length_{D_j}(D_j)\,[D_j/\mathfrak m_j:H],
 \qquad D=\prod_jD_j.
\]
\end{corollary}
Full selected proof · exact TeX
\begin{proof}
A $K$-basis of $C$ becomes an $L$-basis after tensoring. Restriction
of scalars then multiplies the dimension by $n$. For the last
formula, use a composition series in each Artinian local factor;
every simple factor is its residue field. This is the same
length calculation as in Proposition~\ref{f9ds:lengthprojection}.
\end{proof}
Standing context 1 · exact TeX
Let $L/K$ be a finite field extension and let $n=[L:K]$.
