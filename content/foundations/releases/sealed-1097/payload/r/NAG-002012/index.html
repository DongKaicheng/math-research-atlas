\begin{theorem}[\bi{An \'etale-domain factorization in rank zero}{秩零时的 \'etale 域分解}]
\label{fnd:rankzeroetaledomainfactorization}
\bi{Let $f:Y\to X$ be a morphism of good analytic spaces, quasi-smooth of differential rank zero at $y$, and put $x=f(y)$. There are an ordinary affinoid neighborhood $U$ of $x$, a finite \'etale morphism $T\to U$, and an ordinary affinoid neighborhood $V$ of $y$ such that $f|_V$ factors as an affinoid-domain inclusion $V\hookrightarrow T$ followed by $T\to U\hookrightarrow X$.}
{设 $f:Y\to X$ 为 good 解析空间之间的态射，在 $y$ 处准光滑且微分秩为零，令 $x=f(y)$。则存在 $x$ 的普通 affinoid 邻域 $U$、有限 \'etale 态射 $T\to U$ 及 $y$ 的普通 affinoid 邻域 $V$，使 $f|_V$ 分解为 affinoid 域包含 $V\hookrightarrow T$ 与 $T\to U\hookrightarrow X$ 的复合。}
\end{theorem}
