Statement · exact TeX
\begin{corollary}[\bi{Descent of equivariant retractions}{等变收缩的下降}]
\label{cand:galoisretractiondescent}
\bi{Let $S\subseteq X_L$ be $G$-invariant and let $H_L:X_L\times[0,1]\to X_L$ be a continuous $G$-equivariant strong deformation retraction onto $S$. There is a unique continuous strong deformation retraction $H:X\times[0,1]\to X$ onto $p(S)$ such that}
{设 $S\subseteq X_L$ 为 $G$-不变子集，且 $H_L:X_L\times[0,1]\to X_L$ 是到 $S$ 的连续 $G$-等变强形变收缩。存在唯一的连续强形变收缩 $H:X\times[0,1]\to X$ 到 $p(S)$，满足}
\[
 H(p(y),t)=p(H_L(y,t)).
\]
\bi{Writing $r_L=H_L(-,1)$ and $r=H(-,1)$, a continuous function $F:X\to\mathbf R$ satisfies $F\circ p=(F\circ p)\circ r_L$ if and only if $F=F\circ r$.}
{记 $r_L=H_L(-,1)$、$r=H(-,1)$。连续函数 $F:X\to\mathbf R$ 满足 $F\circ p=(F\circ p)\circ r_L$，当且仅当 $F=F\circ r$。}
\end{corollary}
Full selected proof · exact TeX
\begin{proof}
\bi{Equivariance makes $p\circ H_L$ constant on every fibre of $p\times\mathrm{id}_{[0,1]}$. The quotient property in Theorem~\ref{cand:galoisanalyticquotient} produces a unique continuous $H$. Its value at $t=0$ is the identity, its value at $t=1$ lies in $p(S)$, and it fixes every point of $p(S)$ throughout the homotopy: choose a lift in $S$ and use the same assertions for $H_L$. Finally $p\circ r_L=r\circ p$ and surjectivity of $p$ prove the equivalence for $F$.}
{等变性说明 $p\circ H_L$ 在 $p\times\mathrm{id}_{[0,1]}$ 的每个纤维上为常数。由 Theorem~\ref{cand:galoisanalyticquotient} 的商性质，得到唯一的连续 $H$。它在 $t=0$ 为恒等映射，在 $t=1$ 的像包含于 $p(S)$，并在整个形变过程中固定 $p(S)$ 的每个点：在 $S$ 中选择一个提升，再使用 $H_L$ 的对应性质即可。最后，由 $p\circ r_L=r\circ p$ 及 $p$ 的满射性，得到关于 $F$ 的等价。}
\end{proof}
