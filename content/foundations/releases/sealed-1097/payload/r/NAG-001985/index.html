\begin{theorem}[\bi{Ordinary reduction on the strict category}
{Strict 范畴上的通常约化}]
\label{fnd:ordinarystrictgermreduction}
\bi{For a strictly $K$-analytic germ, with $K$ nontrivially
valued, there is a canonical ordinary birational reduction
$\mathscr R_0(X,x)$ with field $\widetilde{\mathcal H(x)}$.
It is functorial and independent of strict affinoid charts
and neighbourhoods. Its nonempty quasi-compact open
subspaces classify strict subdomain germs.
It is related to the graded reduction by}
{对非平凡赋值 $K$ 上的 strict 解析芽，存在具有域
$\widetilde{\mathcal H(x)}$ 的典范通常 birational 约化
$\mathscr R_0(X,x)$。它具有函子性，且与 strict affinoid
图及邻域无关；其非空拟紧开子空间分类 strict 子域芽。
它与分次约化的关系为}
\[
 \mathscr R(X,x)=
 \mathscr R_0(X,x)
   \times_{\mathscr P_{\widetilde K}
                       (\widetilde{\mathcal H(x)})}
   \mathscr P_{\widetilde K^{\rm gr}}
                       (\widetilde{\mathcal H(x)}^{\rm gr}).
\]
\bi{Goodness, separation of germ morphisms, and relative
interiority are detected by affineness, separatedness,
and properness of this ordinary reduction, respectively.}
{Goodness、芽态射的分离性及相对内点性，分别由这个通常
约化的仿射性、分离性及适当性检测。}
\end{theorem}
