\begin{proposition}[Analytic uniform logarithmic excision; F8LA-A1R]
\label[proposition]{prop:analytic}
Assume \cref{hyp:graph}. Let \(\alpha\) be a smooth ambient
\((n,n)\)-form near \(B\), and suppose its ordinary pullback to \(V\)
extends by zero to a compactly supported section \(\alpha_X\) on \(X\),
with support in \(\Int_{\rm top}(V,X)\).

Choose any smooth \(\theta\) as in the proof of \cref{thm:weak}, and let
\[
 u_R(x)=
 \begin{cases}
   \log|f(x)|\,\theta(t_f(x)-R),&x\in V\setminus Z,\\
   0,&x\in Z.
 \end{cases}
 \tag{4.3}
\]
Then \(u_R\) is an ordinary smooth function on \(V\), and
\[
 L_R(\alpha_X):=\int_X\operatorname{ext}_0(u_R\alpha_X|_V)
      =I_C\bigl((-t)\theta(t-R)\pi^*\alpha\bigr).
 \tag{4.4}
\]
For all \(R\geq H\) these numbers equal the single finite value
\[
 L_f(\alpha_X)=I_C((-t)\pi^*\alpha),
 \tag{4.5}
\]
and
\[
 |L_f(\alpha_X)|
 \leq A_C\max\{|a|,|H|\}\,
           \max_{I,J}\sup_B|\alpha_{I,J}|.
 \tag{4.6}
\]
The threshold and constant are independent of the variable coefficients
of \(\alpha\). No boundary-separation or balancing condition is required
for this estimate.
\end{proposition}
