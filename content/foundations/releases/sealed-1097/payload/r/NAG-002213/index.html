\begin{lemma}[\bi{The unique stratum along a face}
{一个面上的唯一分层}]
\label{cand:uniquestratumfacegenerization}
\bi{Let $\eta\in Q_I$ and let $\varnothing\ne J\subseteq I$.
There is exactly one $\eta_J\in Q_J$ whose closure
contains $\eta$. For $J'\subseteq J$, the generization of
$\eta_J$ labelled by $J'$ is $\eta_{J'}$.}
{设 $\eta\in Q_I$，并设 $\varnothing\ne J\subseteq I$。
恰有一个 $\eta_J\in Q_J$ 的闭包包含 $\eta$。
若 $J'\subseteq J$，则 $\eta_J$ 的由 $J'$ 标记的
一般化就是 $\eta_{J'}$。}
\end{lemma}
