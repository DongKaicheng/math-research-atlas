\begin{proposition}[\bi{Formalizing a finite-type analytic covering}
{解析覆盖的形式化}]
\label{fnd:formalcoverrealization}
\bi{Let $\mathfrak X$ be quasi-paracompact and admissible,
and let $(U_\alpha)$ be a finite-type admissible covering of
$\mathfrak X_\eta$ by quasi-compact strict analytic opens.
There is an admissible blowup $\mathfrak X'\to\mathfrak X$
and a finite-type covering by quasi-compact formal opens
$(\mathfrak U_\alpha')$ with generic fibres exactly $U_\alpha$.}
{设 $\mathfrak X$ 为 quasi-paracompact admissible 形式概形，
$(U_\alpha)$ 为 $\mathfrak X_\eta$ 的有限型 admissible 覆盖，
其成员为拟紧 strict 解析开集。存在 admissible blowup
$\mathfrak X'\to\mathfrak X$ 及有限型拟紧形式开覆盖
$(\mathfrak U_\alpha')$，使其泛纤维恰为各 $U_\alpha$。}
\end{proposition}
