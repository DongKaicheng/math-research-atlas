Historical statement · exact TeX
\begin{lemma}[\bi{The first residual coefficient determines the slope}
{首个非零剩余系数确定斜率}]
\label{cand:boundarycoefficientorder}
\bi{Let $g(z)=\sum_{j\geq0}b_jz^j$ on $|z|<1$, with
$b_j\in O$, and suppose its reduction as a formal series is
nonzero of order $d$. Then for all $\rho<1$ sufficiently close
to one,}
{设 $g(z)=\sum_{j\geq0}b_jz^j$ 定义在 $|z|<1$ 上，
$b_j\in O$，且其形式级数约化非零、阶数为 $d$。则对所有
充分接近一的 $\rho<1$，都有}
\[
 |g(\zeta_{0,\rho})|=\rho^d.
\]
\bi{The same conclusion holds near the outer end of
$|a|<|s|<1$ for a bounded double series
$g=\sum_{i,j\geq0}c_{ij}s^i(a/s)^j$, if
$\sum_i\overline c_{i0}s^i$ is nonzero of order $d$.}
{若有界双重级数
$g=\sum_{i,j\geq0}c_{ij}s^i(a/s)^j$ 满足
$\sum_i\overline c_{i0}s^i$ 非零且阶数为 $d$，则同一
结论在 $|a|<|s|<1$ 的外端附近成立。}
\end{lemma}
