\begin{theorem}[\bi{Lifting a global analytic morphism}{全局解析态射的提升}]
\label{fnd:globalformalmorphismlifting}
\bi{Let $\mathfrak X,\mathfrak Y$ be quasi-paracompact
admissible formal schemes. Every analytic morphism
$f:\mathfrak X_\eta\to\mathfrak Y_\eta$ lifts to a formal
morphism $f':\mathfrak X'\to\mathfrak Y$ after a finite
composition $s:\mathfrak X'\to\mathfrak X$ of admissible
blowups, with $f'_\eta=f\circ s_\eta$. Two stages suffice.}
{设 $\mathfrak X,\mathfrak Y$ 为 quasi-paracompact admissible
形式概形。每个解析态射 $f:\mathfrak X_\eta\to\mathfrak Y_\eta$
都在有限次 admissible blowup 的复合
$s:\mathfrak X'\to\mathfrak X$ 后提升为形式态射
$f':\mathfrak X'\to\mathfrak Y$，满足 $f'_\eta=f\circ s_\eta$。
两步即足够。}
\end{theorem}
