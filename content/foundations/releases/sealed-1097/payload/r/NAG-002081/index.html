\begin{proposition}[\bi{The tropical fibre algebra}{热带纤维代数}]
\label{cand:torusfibrealgebra}
\bi{Let $I\subset K[M]$ be an ideal and $X=\operatorname{Spec}(K[M]/I)$. For every real weight $w$, the fibre $X_w=\{x\in X^{\an}:-\log|\chi^{e_i}(x)|=w_i\}$ is the affinoid space}
{设 $I\subset K[M]$ 为理想，$X=\operatorname{Spec}(K[M]/I)$。对任意实权重 $w$，纤维 $X_w=\{x\in X^{\an}:-\log|\chi^{e_i}(x)|=w_i\}$ 是 affinoid 空间}
\[
 X_w=\mathcal M(\mathcal T_w/I\mathcal T_w).
\]
\bi{The ideal $I\mathcal T_w$ is closed and is the norm closure of $I$ in $\mathcal T_w$.}
{理想 $I\mathcal T_w$ 闭，且等于 $I$ 在 $\mathcal T_w$ 中的范数闭包。}
\end{proposition}
Full selected proof · exact TeX
\begin{proof}
\bi{Theorem~\ref{fnd:generalizedfinitemodules} makes every ideal of $\mathcal T_w$ closed. Write an element of $I\mathcal T_w$ as a finite sum $\sum_j h_j f_j$ with $f_j\in I$ and $h_j\in\mathcal T_w$. Approximating the finitely many $h_j$ by Laurent polynomials approximates this sum by elements of $I$. Thus its closure is precisely $I\mathcal T_w$.}
{由 Theorem~\ref{fnd:generalizedfinitemodules}，$\mathcal T_w$ 的每个理想均闭。将 $I\mathcal T_w$ 中的元素写成有限和 $\sum_jh_jf_j$，其中 $f_j\in I$、$h_j\in\mathcal T_w$。用 Laurent 多项式逼近有限个 $h_j$，便以 $I$ 中的元素逼近此和。因此闭包恰为 $I\mathcal T_w$。}

\bi{An analytic point in the indicated fibre bounds every Laurent polynomial by $\|\cdot\|_w$, so its seminorm extends to $\mathcal T_w$ and kills $I\mathcal T_w$. Conversely, the bounds on $T_i,S_i$ and the identities $T_iS_i=1$ force $|T_i|=r_i$ at every point of the displayed quotient. Restriction to $K[M]/I$ gives the corresponding point of $X^{\an}$, by Proposition~\ref{fnd:affineanalytificationpoints}. The two constructions are inverse, continuous in the evaluation topologies, and identify the analytic structures by the rational-domain universal property of Theorem~\ref{fnd:generalizedrationaluniversal}, applied to the finitely many coordinate and inverse-coordinate inequalities.}
{所述纤维中的解析点将每个 Laurent 多项式的值控制在 $\|\cdot\|_w$ 以下，故其半范数延拓到 $\mathcal T_w$ 并消去 $I\mathcal T_w$。反过来，在所列商代数的每个点，$T_i,S_i$ 的界及 $T_iS_i=1$ 强制 $|T_i|=r_i$。由 Proposition~\ref{fnd:affineanalytificationpoints}，限制到 $K[M]/I$ 后给出 $X^{\an}$ 的相应点。两个构造互逆，在取值拓扑下连续。对有限个坐标及逆坐标不等式应用 Theorem~\ref{fnd:generalizedrationaluniversal} 的 rational 域泛性质，便识别解析结构。}
\end{proof}

\section{\bi{Initial ideals and analytic tropical fibres}{初始理想与解析热带纤维}}
\label{cand:initialfibressection}

\bi{Let $K$ be a complete non-Archimedean field, put $v=-\log|\cdot|$, and write $M=\mathbf Z^n$. For $w\in\mathbf R^n$ and $f=\sum_m a_m\chi^m\in K[M]$, put}
{设 $K$ 为完备非阿基米德域，记 $v=-\log|\cdot|$、$M=\mathbf Z^n$。对 $w\in\mathbf R^n$ 及 $f=\sum_m a_m\chi^m\in K[M]$，置}
\[
 \|f\|_w=\max_m |a_m|e^{-\langle m,w\rangle},
 \qquad
 \mu_w(f)=\min_{a_m\ne0}\bigl(v(a_m)+\langle m,w\rangle\bigr).
\]
\bi{A term is active if it attains this maximum, or equivalently this minimum. Denote by $\mathcal T_w$ the completion of $K[M]$ for $\|\cdot\|_w$. The multilaurent presentation of Theorem~\ref{fnd:multilaurentactualpresentation} identifies it, as a normed algebra, with}
{达到上述最大值（等价地，最小值）的项称为活跃项。记 $\mathcal T_w$ 为 $K[M]$ 关于 $\|\cdot\|_w$ 的完备化。由 Theorem~\ref{fnd:multilaurentactualpresentation} 的多重 Laurent 表示，它作为赋范代数等于}
\[
 K\{r_1^{-1}T_1,r_1S_1,\ldots,r_n^{-1}T_n,r_nS_n\}/(T_iS_i-1)_i,
 \qquad r_i=e^{-w_i}.
\]
