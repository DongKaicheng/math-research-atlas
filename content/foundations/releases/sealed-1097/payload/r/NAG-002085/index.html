\begin{theorem}[\bi{Initial degenerations at arbitrary real weights}{任意实权重处的初始退化}]
\label{cand:realinitialdegeneration}
\bi{Choose a complete isometric extension $L/K$ and $t\in(L^\times)^n$ with $v(t)=w$. Such a choice exists for every real $w$. Write $I_t$ for the image of $I L[M]$ under $\chi^m\mapsto t^m\chi^m$, and set}
{选择完备等距扩域 $L/K$ 及 $t\in(L^\times)^n$，使 $v(t)=w$；对每个实 $w$ 均可作此选择。以 $I_t$ 记 $I L[M]$ 在 $\chi^m\mapsto t^m\chi^m$ 下的像，置}
\[
 \operatorname{in}_{w,t,L}(X)
 =\operatorname{Spec}\left(
 \widetilde L[M]\big/\overline{I_t\cap L^\circ[M]}\right).
\]
\bi{After residue-field extension and translation by a point of the residue torus, this scheme is independent of $L,t$. It is nonempty exactly when $w\in\operatorname{Trop}(X)$. For every complete isometric extension $E/K$,}
{在允许剩余域扩张及剩余环面点的平移后，此概形与 $L,t$ 无关。它非空当且仅当 $w\in\operatorname{Trop}(X)$。对每个完备等距扩域 $E/K$，有}
\[
 \operatorname{Trop}(X_E)=\operatorname{Trop}(X).
\]
\end{theorem}
Full selected proof · exact TeX
\begin{proof}
\bi{Extend the multiplicative weighted Gauss norm on $K[M]$ to its fraction field and complete. The coordinate monomials give the required $t$, with the original absolute value on $K$. Multiplicativity is the Gauss calculation of Proposition~\ref{fnd:gaussnorm}. Theorem~\ref{fnd:completevaluedamalgamation} places any two choices in a common complete valued extension. Proposition~\ref{cand:unitinitialbasechange} compares the resulting initial ideals under that extension. For two translating points in the common field, the ratios $t'_i/t_i$ have absolute value one; their residues induce a torus translation between the two special fibres.}
{将 $K[M]$ 上乘法性的加权 Gauss 范数延拓到分式域后完备化，坐标单项式即给出所需 $t$，且 $K$ 上绝对值不变。乘法性由 Proposition~\ref{fnd:gaussnorm} 的 Gauss 计算给出。Theorem~\ref{fnd:completevaluedamalgamation} 将任意两个选择置于共同完备赋值扩域中；Proposition~\ref{cand:unitinitialbasechange} 比较扩域后的初始理想。共同域中的两个平移点之比 $t'_i/t_i$ 的绝对值为一，其剩余类给出两个特殊纤维之间的环面平移。}

\bi{Theorem~\ref{fnd:scalarextensionspectralsurjective}, applied to the fibre algebra of Proposition~\ref{cand:torusfibrealgebra}, shows that every tropical fibre is nonempty before a complete extension exactly when it is nonempty afterwards. This proves tropical base-change invariance. Translation by $t$ identifies the fibre at $w$ with the unit-radius fibre. Proposition~\ref{cand:unitinitialbasechange} and Theorem~\ref{cand:tropicalfundamental} now give the assertion about nonemptiness.}
{对 Proposition~\ref{cand:torusfibrealgebra} 中的纤维代数应用 Theorem~\ref{fnd:scalarextensionspectralsurjective}，可知每个热带纤维在完备扩域前后同时非空。这证明热带化的基变换不变性。再以 $t$ 平移，将 $w$ 处纤维识别为单位半径纤维；由 Proposition~\ref{cand:unitinitialbasechange} 及 Theorem~\ref{cand:tropicalfundamental} 得到非空性的断言。}
\end{proof}

\section{\bi{Initial ideals and analytic tropical fibres}{初始理想与解析热带纤维}}
\label{cand:initialfibressection}

\bi{Let $K$ be a complete non-Archimedean field, put $v=-\log|\cdot|$, and write $M=\mathbf Z^n$. For $w\in\mathbf R^n$ and $f=\sum_m a_m\chi^m\in K[M]$, put}
{设 $K$ 为完备非阿基米德域，记 $v=-\log|\cdot|$、$M=\mathbf Z^n$。对 $w\in\mathbf R^n$ 及 $f=\sum_m a_m\chi^m\in K[M]$，置}
\[
 \|f\|_w=\max_m |a_m|e^{-\langle m,w\rangle},
 \qquad
 \mu_w(f)=\min_{a_m\ne0}\bigl(v(a_m)+\langle m,w\rangle\bigr).
\]
\bi{A term is active if it attains this maximum, or equivalently this minimum. Denote by $\mathcal T_w$ the completion of $K[M]$ for $\|\cdot\|_w$. The multilaurent presentation of Theorem~\ref{fnd:multilaurentactualpresentation} identifies it, as a normed algebra, with}
{达到上述最大值（等价地，最小值）的项称为活跃项。记 $\mathcal T_w$ 为 $K[M]$ 关于 $\|\cdot\|_w$ 的完备化。由 Theorem~\ref{fnd:multilaurentactualpresentation} 的多重 Laurent 表示，它作为赋范代数等于}
\[
 K\{r_1^{-1}T_1,r_1S_1,\ldots,r_n^{-1}T_n,r_nS_n\}/(T_iS_i-1)_i,
 \qquad r_i=e^{-w_i}.
\]
