\begin{proposition}[Native measure decomposition and total variation; F8CR-V]
\label{prop:variation}
For the data of \eqref{eq:components} and a compact ordinary top form
\(\omega\), write \(\omega_i=i_i^*\omega\). Then
\begin{align}
 \mu_{X,\omega}&=\sum_i m_i(i_i)_*\mu_{X_i,\omega_i},
                                      \label{eq:measure-split}\\
 |\mu_{X,\omega}|&\leq\sum_i m_i(i_i)_*|\mu_{X_i,\omega_i}|.
                                      \label{eq:variation}
\end{align}
For each closed embedding separately,
\( |(i_i)_*\nu|=(i_i)_*|\nu|\). In general no equality of the total
variation of the sum is asserted.
\end{proposition}
