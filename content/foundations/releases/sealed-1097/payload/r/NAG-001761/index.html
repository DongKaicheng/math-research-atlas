\begin{proposition}[\bi{A uniform separable regular locus}{统一的可分正则开集}]
\label{fnd:separablenormalizationregularlocus}
\bi{Let $D$ be the affinoid algebra of an affinoid domain in $\mathbb A_K^{n,\an}$, assume that $D$ is a domain, and let $D\hookrightarrow A$ be finite with $A$ a domain. Suppose $\operatorname{Frac}(A)/\operatorname{Frac}(D)$ is separable. There is $0\ne d\in D$ such that, for every complete $L/K$, writing $A_L=A\widehat\otimes_KL$ and $X_L=\mathcal M(A_L)$, the rings}
{设 $D$ 为 $\mathbb A_K^{n,\an}$ 中某个 affinoid 域的代数，并假设 $D$ 为整环。设 $D\hookrightarrow A$ 有限且 $A$ 为整环，并假设 $\operatorname{Frac}(A)/\operatorname{Frac}(D)$ 可分。则存在 $0\ne d\in D$，使得对每个完备扩域 $L/K$，记 $A_L=A\widehat\otimes_KL$、$X_L=\mathcal M(A_L)$ 后，以下各环都正则：}
\[
 A_L[1/d],\qquad
 \mathcal O_{X_L,y}[1/d],\qquad
 \widehat{\mathcal O_{X_L,y}}[1/d]\quad(y\in X_L).
\]
\bi{Here a hat denotes completion at the maximal ideal of the ordinary local ring. In particular this conclusion holds for every strict integral $K$-affinoid algebra $A$ when $\operatorname{char}K=0$, by choosing a finite Tate normalization.}
{这里的帽号指通常局部环在其极大理想处的完备化。特别地，当 $\operatorname{char}K=0$ 时，对每个 strict 整 $K$-affinoid 代数 $A$，选择有限 Tate 正规化即可得到此结论。}
\end{proposition}
