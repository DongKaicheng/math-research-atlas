\begin{theorem}[\bi{The general graded finite-map criterion}
{一般分次有限映射判据}]
\label{fnd:generalgradedfinitecriterion}
\bi{A bounded homomorphism $A\to B$ of generalized affinoid
algebras is finite and admissible if and only if}
{广义 affinoid 代数的有界同态 $A\to B$ 有限且 admissible
当且仅当}
\[
 \widetilde A^{\rm gr}\longrightarrow\widetilde B^{\rm gr}
\]
\bi{is finite. In particular $\widetilde A^{\rm gr}$ is a
finitely generated graded $\widetilde K^{\rm gr}$-algebra.}
{为有限映射。特别地，$\widetilde A^{\rm gr}$ 是有限生成的
分次 $\widetilde K^{\rm gr}$-代数。}
\end{theorem}
