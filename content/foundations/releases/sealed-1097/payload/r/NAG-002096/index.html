\begin{lemma}[\bi{Galois rationalization of a residue-point support}
{剩余点支撑的 Galois 有理化}]
\label{cand:charzeroresiduepointsplitting}
\bi{Assume $\operatorname{char}K=0$. For every finite
extension $\kappa/k$, there exists a finite Galois
extension $N/K$ such that all points of
$\operatorname{Spec}(\kappa\otimes_k\widetilde N)$ are
$\widetilde N$-rational and
$\operatorname{Gal}(N/K)$ acts transitively on this finite
set of points.}
{假设 $\operatorname{char}K=0$。对每个有限扩张
$\kappa/k$，存在有限 Galois 扩张 $N/K$，使
$\operatorname{Spec}(\kappa\otimes_k\widetilde N)$ 的
全部点都是 $\widetilde N$-有理点，并且
$\operatorname{Gal}(N/K)$ 在这个有限点集上传递作用。}
\end{lemma}
Full selected proof · exact TeX
\begin{proof}
\bi{By \cite[Tag 030K]{StacksAlg}, write
$\kappa/\kappa_s/k$ with $\kappa_s/k$ separable and
$\kappa/\kappa_s$ purely inseparable.
Choose a finite Galois extension $l_s/k$ containing the
normal closure of $\kappa_s/k$. There is a finite purely
inseparable extension $l/l_s$ containing every conjugate
of $\kappa$ over $k$: for finitely many generators $\beta_i$
of $\kappa$, choose powers $\beta_i^{p^{e_i}}\in\kappa_s$,
and adjoin to $l_s$ the unique $p^{e_i}$-th roots of their
images under all embeddings $\kappa_s\hookrightarrow l_s$.
When $\operatorname{char}k=0$, take $l=l_s$.
The field $l$ therefore contains the image of every
$k$-embedding of $\kappa$ into an algebraic closure.}
{由 \cite[Tag 030K]{StacksAlg}，可写成
$\kappa/\kappa_s/k$，其中 $\kappa_s/k$ 可分，而
$\kappa/\kappa_s$ 纯不可分。选取包含
$\kappa_s/k$ 正规闭包的有限 Galois 扩张 $l_s/k$。
存在包含 $\kappa$ 的全部 $k$-共轭的有限纯不可分
扩张 $l/l_s$：对 $\kappa$ 的有限个生成元 $\beta_i$，
取满足 $\beta_i^{p^{e_i}}\in\kappa_s$ 的幂，再对全部
嵌入 $\kappa_s\hookrightarrow l_s$，向 $l_s$ 添入这些
像的唯一 $p^{e_i}$ 次根。当 $\operatorname{char}k=0$
时，取 $l=l_s$。因此 $l$ 包含 $\kappa$ 到代数闭包
的每个 $k$-嵌入的像。}

\bi{Lift $l_s/k$ to a finite Galois valued extension $E/K$
by Lemma~\ref{cand:galoisresidueextensionlift}. Over $E$,
apply Lemma~\ref{cand:arbitraryresidueextensionlift} to
$l/l_s$, obtaining $F/E$ with residue field $l$.
Since $K$ has characteristic zero, $F/K$ is separable.
Adjoining the finitely many roots of minimal polynomials
of generators of $F/K$ gives a finite normal extension
$N/K$ containing $F$, by \cite[Tag 09HU]{StacksAlg}.
It is Galois, and $\widetilde N$ contains $l$.}
{由 Lemma~\ref{cand:galoisresidueextensionlift} 将
$l_s/k$ 提升为有限 Galois 赋值扩张 $E/K$。在 $E$
上，对 $l/l_s$ 应用
Lemma~\ref{cand:arbitraryresidueextensionlift}，得到
剩余域为 $l$ 的 $F/E$。由于 $K$ 特征零，$F/K$
可分。添入 $F/K$ 的有限个生成元的极小多项式的全部
根，由 \cite[Tag 09HU]{StacksAlg} 得到包含 $F$ 的有限
正规扩张 $N/K$。它是 Galois 扩张，且
$\widetilde N$ 包含 $l$。}

\bi{Any residue field of the finite $\widetilde N$-algebra
$\kappa\otimes_k\widetilde N$ embeds in an algebraic
closure of $\widetilde N$. Its image is generated by
$\widetilde N$ and a $k$-embedded copy of $\kappa$.
The latter is already in $l$, so that residue field is
$\widetilde N$. Thus the points are precisely the distinct
$k$-embeddings of $\kappa$ into $\widetilde N$.
Restriction from $\operatorname{Gal}(N/K)$ onto
$\operatorname{Gal}(E/K)$ is surjective by
\cite[Tag 0BME]{StacksAlg}; the latter group induces
$\operatorname{Gal}(l_s/k)$. It acts transitively on the
embeddings of $\kappa_s$, and a purely inseparable extension
has at most one extension of any embedding. All embeddings
of $\kappa$ exist in $l$, so this proves transitivity.}
{有限 $\widetilde N$-代数
$\kappa\otimes_k\widetilde N$ 的任意剩余域都可嵌入
$\widetilde N$ 的代数闭包，其像由 $\widetilde N$ 和
$\kappa$ 的一个 $k$-嵌入像生成。后者已经包含于 $l$，
故该剩余域就是 $\widetilde N$。因此其点恰为
$\kappa$ 到 $\widetilde N$ 的不同 $k$-嵌入。
由 \cite[Tag 0BME]{StacksAlg}，
$\operatorname{Gal}(N/K)$ 到
$\operatorname{Gal}(E/K)$ 的限制满射，而后者在
剩余域上诱导 $\operatorname{Gal}(l_s/k)$。这个群
在 $\kappa_s$ 的嵌入上传递作用；每个嵌入向纯不可分
扩张至多有一个延伸。$\kappa$ 的全部嵌入都存在于
$l$ 中，因此得到传递性。}
\end{proof}
