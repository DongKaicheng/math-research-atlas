\begin{theorem}[\bi{Goodness is detected by graded reduction}
{分次约化检测 goodness}]
\label{fnd:goodgermiffaffinereduction}
\bi{An analytic germ over an arbitrary complete non-Archimedean
field is good if and only if its graded birational reduction
is affine.}
{任意完备非阿基米德域上的解析芽为 good，当且仅当其分次
birational 约化仿射。}
\end{theorem}
