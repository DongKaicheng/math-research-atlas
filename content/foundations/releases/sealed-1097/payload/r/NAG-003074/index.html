\begin{lemma}[Compact faithfulness and ordinary presentation]\label{lem:cld}
Let $C\subset X^\an$ be affinoid, let $f$ be a finite analytic unit tuple on $C$, and let $\alpha$ be a smooth ambient superform near $s_f(C)$. Then
\begin{equation}\label{eq:compactkernel}
 s_f^*\alpha=0\text{ in the CLD G-sheaf on }C
       \quad\Longleftrightarrow\quad \alpha|_{s_f(C)}=0.
\end{equation}
Every germ of $\A_{\rm CLD}$ has a presentation by a finite analytic unit tuple on an ordinary open neighborhood and an ambient smooth superform. Consequently \eqref{eq:cldmap} defines an isomorphism of ordinary sheaves
\[
                           c:\A_{\rm mom}\xrightarrow{\sim}\A_{\rm CLD}.
\]
\end{lemma}
