\begin{theorem}[\bi{Inner flatness with nontrivially valued completed residue field}{完备剩余域非平凡赋值时的内点平坦性}]
\label{fnd:nontrivialinnercoherentflatness}
\bi{Let $f:Y\to X$ be a morphism of good analytic spaces over any complete field and let $\mathcal F$ be coherent on $Y_G$. At an inner point $y$, with $\mathcal H(f(y))$ nontrivially valued, naive $X$-flatness of $\mathcal F$ is equivalent to analytic $X$-flatness.}
{设 $f:Y\to X$ 为任意完备域上 good 解析空间之间的态射，$\mathcal F$ 在 $Y_G$ 上凝聚。在每个满足 $\mathcal H(f(y))$ 非平凡赋值的内点 $y$ 处，$\mathcal F$ 的朴素 $X$-平坦性等价于解析 $X$-平坦性。}
\end{theorem}
