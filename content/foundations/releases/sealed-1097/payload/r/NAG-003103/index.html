\begin{theorem}[Non-\(S_1\) fixed-test identities; F8NM-G]\label{thm:fixed}
Under the exact reviewed component/scalar and ZP inputs above, every
ambient regular meromorphic \(f\) on \(U\) satisfies
\begin{equation}\label{eq:meromorphic}
 L_f(\dd\eta)=\delta_{\divis(f)}(\eta)
 \qquad(\eta\in\A_c^{n-1,n-1}(U)).
\end{equation}
Let \(\overline{\cL}\) have ordinary smooth metric weights and a given
ambient regular meromorphic section \(s\). Its potential pairing
\(G_s\), full Cartier divisor \(D_s\), and ordinary curvature satisfy
\begin{equation}\label{eq:metric}
 G_s(\dd\eta)+\delta_{D_s}(\eta)
       =\int_U c_1(\overline{\cL})\wedge\eta.
\end{equation}
These are fixed-test statements before any topology is invoked.
\end{theorem}
