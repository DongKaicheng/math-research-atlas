\begin{theorem}[\bi{Ordinary regularity of quasi-smooth morphisms}{准光滑态射的通常正则性}]
\label{fnd:quasismoothordinaryregular}
\bi{Let $f:Y\to X$ be a morphism of good analytic spaces over any complete non-Archimedean field, quasi-smooth at $y$, and put $x=f(y)$. Then $\mathcal O_{X,x}\to\mathcal O_{Y,y}$ is a regular homomorphism of Noetherian local rings. If $X=\mathcal M(A)$ and $Y=\mathcal M(B)$ are affinoid, then $A_{\ker(x)}\to B_{\ker(y)}$ is regular as well.}
{设 $f:Y\to X$ 为任意完备非阿基米德域上 good 解析空间之间的态射，在 $y$ 处准光滑，并令 $x=f(y)$。则 $\mathcal O_{X,x}\to\mathcal O_{Y,y}$ 是 Noetherian 局部环之间的正则同态。若 $X=\mathcal M(A)$、$Y=\mathcal M(B)$ 为 affinoid，则 $A_{\ker(x)}\to B_{\ker(y)}$ 也正则。}
\end{theorem}
