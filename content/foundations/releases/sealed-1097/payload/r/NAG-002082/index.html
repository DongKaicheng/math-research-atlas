\begin{theorem}[\bi{The tropical fundamental theorem over the original field}{原基域上的热带基本定理}]
\label{cand:tropicalfundamental}
\bi{With the preceding notation, the following conditions on $w\in\mathbf R^n$ are equivalent:}
{沿用上述记号，对 $w\in\mathbf R^n$，下列条件等价：}
\begin{enumerate}
\item $w\in\operatorname{Trop}(X):=\operatorname{trop}(X^{\an})$;
\item $\mathcal T_w/I\mathcal T_w\ne0$;
\item \bi{every nonzero $f\in I$ has at least two active terms at $w$.}{每个非零 $f\in I$ 在 $w$ 处至少有两个活跃项。}
\end{enumerate}
\bi{This holds with trivial valuation, arbitrary real radii, and nonreduced $X$.}
{此结论允许平凡赋值、任意实半径及非约化 $X$。}
\end{theorem}
Full selected proof · exact TeX
\begin{proof}
\bi{The equivalence of the first two conditions follows from Proposition~\ref{cand:torusfibrealgebra} and Theorem~\ref{fnd:generalizedpointunitcriterion}. If $f$ has a unique active term $a_m\chi^m$, then}
{前两项的等价由 Proposition~\ref{cand:torusfibrealgebra} 及 Theorem~\ref{fnd:generalizedpointunitcriterion} 给出。若 $f$ 的唯一活跃项为 $a_m\chi^m$，则}
\[
 f=a_m\chi^m(1+h),\qquad \|h\|_w<1.
\]
\bi{Lemma~\ref{fnd:geometricinverse} makes $f$ invertible in $\mathcal T_w$, so the quotient is zero. Conversely, if that quotient is zero, then $1\in I\mathcal T_w$. By the closure assertion in Proposition~\ref{cand:torusfibrealgebra}, there is $g\in I$ with $\|1-g\|_w<1$. Its constant coefficient has absolute value one and every other term has weighted size less than one. Thus $g$ has a unique active term, contradicting the third condition.}
{由 Lemma~\ref{fnd:geometricinverse}，$f$ 在 $\mathcal T_w$ 中可逆，故商为零。反之，若商为零，则 $1\in I\mathcal T_w$。由 Proposition~\ref{cand:torusfibrealgebra} 的闭包断言，存在 $g\in I$ 使 $\|1-g\|_w<1$。其常数项的绝对值为一，其余每项的加权大小小于一，所以 $g$ 有唯一活跃项，与第三项矛盾。}
\end{proof}

\section{\bi{Initial ideals and analytic tropical fibres}{初始理想与解析热带纤维}}
\label{cand:initialfibressection}

\bi{Let $K$ be a complete non-Archimedean field, put $v=-\log|\cdot|$, and write $M=\mathbf Z^n$. For $w\in\mathbf R^n$ and $f=\sum_m a_m\chi^m\in K[M]$, put}
{设 $K$ 为完备非阿基米德域，记 $v=-\log|\cdot|$、$M=\mathbf Z^n$。对 $w\in\mathbf R^n$ 及 $f=\sum_m a_m\chi^m\in K[M]$，置}
\[
 \|f\|_w=\max_m |a_m|e^{-\langle m,w\rangle},
 \qquad
 \mu_w(f)=\min_{a_m\ne0}\bigl(v(a_m)+\langle m,w\rangle\bigr).
\]
\bi{A term is active if it attains this maximum, or equivalently this minimum. Denote by $\mathcal T_w$ the completion of $K[M]$ for $\|\cdot\|_w$. The multilaurent presentation of Theorem~\ref{fnd:multilaurentactualpresentation} identifies it, as a normed algebra, with}
{达到上述最大值（等价地，最小值）的项称为活跃项。记 $\mathcal T_w$ 为 $K[M]$ 关于 $\|\cdot\|_w$ 的完备化。由 Theorem~\ref{fnd:multilaurentactualpresentation} 的多重 Laurent 表示，它作为赋范代数等于}
\[
 K\{r_1^{-1}T_1,r_1S_1,\ldots,r_n^{-1}T_n,r_nS_n\}/(T_iS_i-1)_i,
 \qquad r_i=e^{-w_i}.
\]
