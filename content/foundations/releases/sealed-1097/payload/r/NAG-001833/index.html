\begin{definition}[\bi{The finite affinoid-domain site}{有限 affinoid 域 site}]
\label{fnd:generalizedfinitedomainsite}
\bi{For $X=\mathcal M(A)$, let $\mathcal D_X$ be the category of all affinoid domains in $X$, with inclusions as arrows. Give it finite set-theoretic coverings by its objects, including the empty covering of the empty domain. Choose representing algebras $A_V$, with the unique bounded restriction maps, and define}
{对 $X=\mathcal M(A)$，令 $\mathcal D_X$ 为 $X$ 中所有 affinoid 域组成的范畴，以包含为箭头。覆盖取由其对象组成的有限集合覆盖，并包括空域的空覆盖。选择各表示代数 $A_V$ 及其唯一有界限制映射，定义}
\[
 \mathcal O_X^{\mathrm{aff}}(V)=A_V,
 \qquad \widetilde M^{\mathrm{aff}}(V)=A_V\otimes_A M
 \quad(M\text{ finite over }A).
\]
\end{definition}
