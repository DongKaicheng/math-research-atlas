\begin{theorem}[\bi{Naive and analytic flatness at every inner point}{每个内点处的朴素与解析平坦性}]
\label{fnd:generalinnercoherentflatness}
\bi{Let $K$ be any complete non-Archimedean field, let $f:Y\to X$ be a morphism of good $K$-analytic spaces, and let $\mathcal F$ be coherent on $Y_G$. For every $y\in\operatorname{Int}(Y/X)$, the following are equivalent: $\mathcal F_y$ is flat over $\mathcal O_{X,f(y)}$; and $\mathcal F$ is analytically $X$-flat at $y$, tested after every good analytic base change over every complete valued extension of $K$.}
{设 $K$ 为任意完备非阿基米德域，$f:Y\to X$ 为 good $K$-解析空间间的态射，$\mathcal F$ 在 $Y_G$ 上凝聚。对每个 $y\in\operatorname{Int}(Y/X)$，下列条件等价：$\mathcal F_y$ 在 $\mathcal O_{X,f(y)}$ 上平坦；以及 $\mathcal F$ 在 $y$ 处解析 $X$-平坦，即在 $K$ 的每个完备赋值扩张上的每个 good 解析基变换后均通过平坦性测试。}
\end{theorem}
