\begin{lemma}[\bi{A closedness criterion}{Closedness criterion}]\label{fnd:controlledclosed}
\bi{Let $R$ be a complete non-Archimedean normed $K$-algebra, let $E$ be a complete non-Archimedean normed $R$-module with $\|av\|\leq\|a\|\|v\|$, and let $N\subset E$ be an algebraic submodule. Suppose that there are $v_1,\ldots,v_s\in N$ and $C>0$ such that every $v\in N$ has an expression}
{设 $R$ 为 complete non-Archimedean normed $K$-algebra，$E$ 为满足 $\|av\|\leq\|a\|\|v\|$ 的 complete non-Archimedean normed $R$-module，$N\subset E$ 为 algebraic submodule。假设存在 $v_1,\ldots,v_s\in N$ 和 $C>0$，使每个 $v\in N$ 都可以表示为}
\[
 v=\sum_{i=1}^s a_i v_i,\qquad \max_i\|a_i\|\leq C\|v\|.
\]
\bi{Then $N$ is closed, and $R^s\to N$ is a bounded, norm-strict surjection for the induced norm on $N$.}
{则 $N$ 为 closed，且对 $N$ 上的 induced norm，$R^s\to N$ 为 bounded、norm-strict surjection。}
\end{lemma}
Full selected proof · exact TeX
\begin{proof}
\bi{If $v_j\in N$ converges in $E$, choose coefficients $a_{ij}$ representing $v_{j+1}-v_j$ with $\|a_{ij}\|\leq C\|v_{j+1}-v_j\|$. These coefficients tend to zero. Each series $\sum_j a_{ij}$ therefore converges in $R$. Express $v_1$ in the same generators and add these coefficient series. Continuity of the finite sum shows that $\lim_jv_j\in N$. The generator map is bounded by $\max_i\|v_i\|$, while its quotient norm is at most $C$ times the norm of its image. These two estimates give norm-strictness.}
{若 $v_j\in N$ 在 $E$ 中收敛，选取表示 $v_{j+1}-v_j$ 的 coefficients $a_{ij}$，满足 $\|a_{ij}\|\leq C\|v_{j+1}-v_j\|$。这些 coefficients 趋于零，故每个 series $\sum_j a_{ij}$ 在 $R$ 中收敛。将 $v_1$ 用同一组 generators 表示，再加上这些 coefficient series。由 finite sum 的 continuity 得到 $\lim_jv_j\in N$。Generator map 的 bound 为 $\max_i\|v_i\|$，而 quotient norm 不超过其 image 的 norm 乘以 $C$。这两个 estimates 给出 norm-strictness。}
\end{proof}

\chapter{\bi{Valued fields, Banach algebras and analytic points}{Valued fields、Banach algebras 与 analytic points}}\label{ch:1}\label{vol:I}\label{part:tropical}\label{fnd:algebras}
\bi{A non-Archimedean analytic space is locally described by convergent functions, but its points need not be evaluations at elements of the ground field. The purpose of this chapter is to construct both parts of this description: the algebras of functions and the valued fields in which their values lie. The next chapter supplies domains and sheaves. This order is important for multiplicities: a seminorm detects the value of a function, whereas the algebra also retains nilpotents and finite-module lengths.}
{Non-Archimedean analytic space 局部由 convergent functions 描述，但其 points 未必来自 ground field 中元素的 evaluation。本章构造这一描述的两部分：functions 的 algebras，以及函数值所在的 valued fields。下一章再建立 domains 与 sheaves。这个顺序对 multiplicities 很重要：seminorm 检测函数值，而 algebra 还保留 nilpotents 与 finite-module lengths。}

\section{\bi{The ground field and convergence}{Ground field 与 convergence}}\label{fnd:valuedfields}
\begin{definition}\label{fnd:fieldconventions}
\bi{Throughout the foundations, $K$ is a field complete for a real-valued non-Archimedean absolute value. We permit the trivial absolute value unless a statement explicitly excludes it. Set}
{基础篇中，$K$ 是关于 real-valued non-Archimedean absolute value 完备的 field。除非明确排除，否则允许 trivial absolute value。记}
\[
\begin{gathered}
v(a)=-\log|a|,\qquad \Gamma=v(K^\times),\\
K^\circ=\{a\in K:|a|\leq1\},\qquad K^{\circ\circ}=\{a\in K:|a|<1\},\\
\widetilde K=K^\circ/K^{\circ\circ}.
\end{gathered}
\]
\bi{A complete extension $L/K$ always carries an absolute value restricting to the specified one on $K$. We do not rescale it when changing the ground field.}

\begin{definition}\label{fnd:banachalgebra}
\bi{A Banach $K$-algebra is a commutative unital $K$-algebra $A$ with a complete non-Archimedean norm satisfying $\|ca\|=|c|\|a\|$, $\|1\|=1$, and $\|ab\|\leq\|a\|\|b\|$. A homomorphism is bounded if $\|\phi(a)\|\leq C\|a\|$ for some constant $C$. Equivalent norms define the same Banach topology. A surjective bounded map is admissible if its target norm is uniformly equivalent to the quotient norm; equivalently, it admits lifts with a uniform norm bound. Equality of topologies alone is insufficient at trivial valuation; see Definition~\ref{fnd:normstrictconvention} and Example~\ref{fnd:normtopologycounterexample}.}
{Banach $K$-algebra 是 commutative unital $K$-algebra $A$，配备一个 complete norm。此 norm 为 non-Archimedean，且满足 $\|ca\|=|c|\|a\|$、$\|1\|=1$ 和 $\|ab\|\leq\|a\|\|b\|$。若存在常数 $C$ 使 $\|\phi(a)\|\leq C\|a\|$，则 homomorphism 称为 bounded。Equivalent norms 定义相同的 Banach topology。Surjective bounded map 若使 target norm 与 quotient norm uniformly equivalent，则称为 admissible；等价地，它具有 uniform norm bound 的 lifts。赋值平凡时，仅有 topologies 相同还不够；见 Definition~\ref{fnd:normstrictconvention} 及 Example~\ref{fnd:normtopologycounterexample}。}
\end{definition}

\begin{definition}\label{fnd:tatereal}
\bi{For $r=(r_1,\ldots,r_n)\in\R_{>0}^n$, the generalized Tate algebra is}
{对 $r=(r_1,\ldots,r_n)\in\R_{>0}^n$，定义 generalized Tate algebra}
\[
 K\{r^{-1}T\}=
 \left\{\sum_{\nu\in\N^n}a_\nu T^\nu:
       |a_\nu|r^\nu\longrightarrow0\text{ as }|\nu|\to\infty\right\}.
\]
\[
 \|f\|_r=\max_\nu |a_\nu|r^\nu.
\]
\bi{Here $r^\nu=\prod_i r_i^{\nu_i}$. When all $r_i=1$, write $K\langle T_1,\ldots,T_n\rangle$.}
{这里 $r^\nu=\prod_i r_i^{\nu_i}$。当所有 $r_i=1$ 时，记作 $K\langle T_1,\ldots,T_n\rangle$。}
\end{definition}

\begin{definition}\label{fnd:affinoiddefinition}
\bi{A $K$-affinoid algebra is a Banach $K$-algebra admitting an admissible surjection $K\{r^{-1}T\}\twoheadrightarrow A$ for some finite tuple of positive radii. It is strictly $K$-affinoid if such a presentation exists with all radii equal to one. Morphisms are bounded $K$-algebra homomorphisms.}
{$K$-affinoid algebra 是一个 Banach $K$-algebra，它对某个有限的 positive-radii tuple 存在 admissible surjection $K\{r^{-1}T\}\twoheadrightarrow A$。若可取所有 radii 等于一，则称为 strictly $K$-affinoid。Morphisms 为 bounded $K$-algebra homomorphisms。}
\end{definition}

\begin{definition}\label{fnd:spectrum}
\bi{For a $K$-affinoid algebra $A$, its Berkovich spectrum $\mathcal M(A)$ is the set of bounded multiplicative seminorms extending $|\cdot|_K$. Its topology is the weakest topology making every evaluation $x\mapsto|a(x)|$ continuous. The algebra $A$, not just this topological space, is part of the affinoid object.}
{对 $K$-affinoid algebra $A$，其 Berkovich spectrum $\mathcal M(A)$ 为延拓 $|\cdot|_K$ 的 bounded multiplicative seminorms 的集合。其 topology 是使所有 evaluation $x\mapsto|a(x)|$ continuous 的最弱 topology。Affinoid object 的数据包括 algebra $A$，而不只是这个 topological space。}
\end{definition}

\section{\bi{Norm conventions and bounded localization}{Norm conventions 与 bounded localization}}\label{fnd:wp01norms}
\bi{Before constructing localization maps, we fix the quantitative meaning of admissibility. This matters when the field is trivially valued. The correction below replaces the topology-only wording in version 0.26; its original wording is retained in the version record. All later uses of ``admissible'' refer to equivalent quotient norms.}
{构造 localization maps 之前，先固定 admissibility 的定量含义。当基域赋值平凡时，这一区别不可省略。下述修正替换 v0.26 中仅用 topology 表述的约定；原措辞保留在 version record 中。此后 ``admissible'' 均指 equivalent quotient norms。}

\begin{definition}\label{fnd:normstrictconvention}
\bi{For a bounded linear map $u:E\to F$, give $\operatorname{im}u$ both its induced norm from $F$ and its quotient seminorm from $E$. The map is norm-strict, or admissible, when these are uniformly equivalent. Since $u$ is bounded, the additional requirement is a constant $C>0$ such that every $y\in\operatorname{im}u$ has a lift $x$ with $\|x\|\leq C\|y\|$. The constant may be enlarged by any fixed factor greater than one to approximate an unattained infimum.}
{对 bounded linear map $u:E\to F$，在 $\operatorname{im}u$ 上同时考虑来自 $F$ 的 induced norm 与来自 $E$ 的 quotient seminorm。当两者 uniformly equivalent 时，称该 map 为 norm-strict，或 admissible。由于 $u$ 已 bounded，额外条件等价于存在常数 $C>0$，使每个 $y\in\operatorname{im}u$ 都有 lift $x$ 满足 $\|x\|\leq C\|y\|$。当 infimum 不取到时，可将常数乘以任意固定的大于一的因子。}
\end{definition}
\source{\cite[Definition/Exercise 2.1.1.1(ii)--(iii), p.~4]{TemIntro}.}

\section{\bi{Controlled generation in strict Tate algebras}{Strict Tate algebras 中带控制的有限生成}}\label{fnd:wp02strict}
\bi{Division over a Banach coefficient algebra supplies an induction in the number of variables. We carry out that induction with estimates: finite generation alone would not prove that an algebraic submodule is closed. Put $R_n=K\langle X_1,\ldots,X_n\rangle$, with the unit Gauss norm, and $R_0=K$. Finite powers carry the maximum norm. The field may be trivially valued. Every constant below is independent of the element being represented.}
{Banach coefficient algebra 上的 division 可以用于按变量个数归纳。下面在归纳中同时保留 estimates，因为仅有 finite generation 并不能证明 algebraic submodule 为 closed。记 $R_n=K\langle X_1,\ldots,X_n\rangle$，配备 unit Gauss norm，并令 $R_0=K$。Finite powers 配备 maximum norm。允许基域的赋值平凡。下文各常数均不依赖所表示的元素。}
