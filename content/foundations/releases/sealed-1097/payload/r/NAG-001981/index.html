\begin{lemma}[\bi{Preparing a two-sided Laurent neighbourhood}
{双侧 Laurent 邻域的准备}]
\label{fnd:gradedlaurentpairpreparation}
\bi{Suppose a separated germ $(X,x)$ is covered by good
subdomain germs $Y,Z$, and its reduction is affine. Suppose,
for a nonzero homogeneous $\lambda\in F_x$, that}
{设分离芽 $(X,x)$ 被 good 子域芽 $Y,Z$ 覆盖，且其约化
仿射。假设对某个非零齐次 $\lambda\in F_x$，有}
\[
 \mathscr R(Y,x)=\mathscr R(X,x)\{\lambda\},\qquad
 \mathscr R(Z,x)=\mathscr R(X,x)\{\lambda^{-1}\}.
\]
\bi{Then one can choose affinoid neighbourhood representatives
$Y=\mathcal M(B)$, $Z=\mathcal M(C)$, whose union is a
neighbourhood of $x$, and elements and radii satisfying every
hypothesis of Theorem~\ref{fnd:doublelaurentaffinoidgluing}.}
{则可以选取 affinoid 邻域代表
$Y=\mathcal M(B)$、$Z=\mathcal M(C)$，使其并为 $x$ 的
邻域，并选取满足
Theorem~\ref{fnd:doublelaurentaffinoidgluing} 全部假设的元素
及半径。}
\end{lemma}
