\begin{proposition}[\bi{An algebraic pointed \texorpdfstring{\'etale}{etale}
map preserves the tube}{代数带点 \texorpdfstring{\'etale}{etale} 态射保持管域}]
\label{cand:algebraicetaletubes}
\bi{Let $f:\mathcal Y\to\mathcal X$ be an \'etale morphism
between flat finitely presented $O$-schemes. Let
$q'\in\mathcal Y_s(k)$ and $q=f(q')\in\mathcal X_s(k)$.
The induced generic-fibre morphism restricts to an analytic
isomorphism}
{设 $f:\mathcal Y\to\mathcal X$ 是平坦有限表示 $O$-概形
之间的 \'etale 态射，且 $q'\in\mathcal Y_s(k)$、
$q=f(q')\in\mathcal X_s(k)$。诱导的泛纤维态射限制为解析同构}
\[
 ]q'[_{\widehat{\mathcal Y}}
       \ \xrightarrow{\ \sim\ }\ ]q[_{\widehat{\mathcal X}}.
\]
\end{proposition}
Full selected proof · exact TeX
\begin{proof}
\bi{Lemma~\ref{cand:affineflatformalcompletion}
makes these completions admissible, and is compatible with formal
open restriction. A Zariski open containing the selected residual
point retains its entire tube. We may therefore replace the source
and target by affine neighbourhoods of $q'$ and $q$.}
{Lemma~\ref{cand:affineflatformalcompletion} 使这些完备化成为 admissible 形式
概形，并与形式开集限制相容。包含所选剩余点的 Zariski 开集
保留该点的整管域。因此可将源与靶分别换为 $q'$、$q$ 的
仿射邻域。}

\bi{By the square Jacobian presentation in
\cite[Tag 02GU]{StacksAlg}, after such shrinking the morphism is
given by $R\to S$, where}
{由 \cite[Tag 02GU]{StacksAlg} 的方阵 Jacobian 表示，作上述
收缩后，态射由 $R\to S$ 给出，其中}
\[
 \begin{gathered}
 S=R[Z_1,\ldots,Z_n]/(F_1,\ldots,F_n),\\
 J(Z)=\left(\frac{\partial F_i}{\partial Z_j}\right)_{i,j},
 \qquad \det J(q')\ne0.
 \end{gathered}
\]
\bi{Localize once more at $\det J$, which is nonzero at $q'$.
Adjoining its inverse as a variable $Q$ gives the presentation
$(F_1,\ldots,F_n,Q\det J-1)$. Completing the polynomial
presentation by Lemma~\ref{cand:affineflatformalcompletion} gives}
{再在 $q'$ 处非零的 $\det J$ 上局部化。以变量 $Q$ 添入
其逆，得到关系 $(F_1,\ldots,F_n,Q\det J-1)$。按
Lemma~\ref{cand:affineflatformalcompletion} 完成该多项式表示，得到}
\[
 \widehat S=\widehat R\langle Z_1,\ldots,Z_n,Q\rangle/
                 (F_1,\ldots,F_n,Q\det J-1).
\]
\bi{Here quotients are the separated completed quotients;
their indicated finite relation ideals are closed by the
admissible-presentation result. Choose lifts
$z_{0i}\in O$ of the coordinates $Z_i(q')\in k$.
They satisfy the hypotheses of
Corollary~\ref{cand:standardetaletubeisomorphism}, which supplies
the required entire-tube isomorphism, including its analytic
structure. The case of no variables is the identity case.
All the preceding restrictions preserved the tubes, completing
the proof.}
{这里商均为分离完备商；所列有限关系理想由 admissible
表示定理为闭理想。选择坐标 $Z_i(q')\in k$ 的提升
$z_{0i}\in O$，它们满足
Corollary~\ref{cand:standardetaletubeisomorphism} 的假设，
故得到包括解析结构在内的整管域同构。没有变量的情形就是
恒等态射。此前全部限制均保留管域，由此完成证明。}
\end{proof}
