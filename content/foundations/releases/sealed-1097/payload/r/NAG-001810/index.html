\begin{theorem}[\bi{Domain locality and analytic base change}{域局部性与解析基变换}]
\label{fnd:generalboundarybasechangeglocal}
\bi{Let $(X_i)$ be an admissible analytic-domain covering of $X$ and put $Y_i=f^{-1}(X_i)$. Then}
{设 $(X_i)$ 为 $X$ 的 admissible 解析域覆盖，并记 $Y_i=f^{-1}(X_i)$。则}
\[
 \partial(Y/X)=\bigcup_i\partial(Y_i/X_i).
\]
\bi{If $X'\to X$ is an analytic morphism over any complete valued extension $L/K$, $Y'=Y\times_X X'$, and $y'\in Y'$ maps to $y\in Y$, then}
{若 $X'\to X$ 是任意完备赋值扩张 $L/K$ 上的解析态射，$Y'=Y\times_X X'$，且 $y'\in Y'$ 映到 $y\in Y$，则}
\[
 y\in\operatorname{Int}(Y/X)
       \quad\Longrightarrow\quad
 y'\in\operatorname{Int}(Y'/X').
\]
\end{theorem}
