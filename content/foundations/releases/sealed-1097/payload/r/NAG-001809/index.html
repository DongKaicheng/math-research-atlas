\begin{theorem}[\bi{Openness of relative interior}{相对内部的开性}]
\label{fnd:generalrelativeinterioropen}
\bi{For every morphism of analytic spaces, $\operatorname{Int}(Y/X)$ is open in the ordinary topology of $Y$.}
{对任意解析空间态射，$\operatorname{Int}(Y/X)$ 在 $Y$ 的普通拓扑中为开集。}
\end{theorem}
