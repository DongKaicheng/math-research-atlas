\begin{proposition}[Fixed-test shell identity; F8F7-R]
\label{prop:shell}
Assume the exact F7 statements used above. There exists \(R_0=R_0(f,\eta)\)
such that, for every \(R\geq R_0\),
\[
 \begin{split}
 I_{\OO_X}\!\left(
   \operatorname{ext}_0(\log|f|\,\dd\eta|_{X\setminus D})\right)
 &= I_{\OO_X}(u_R\,\dd\eta)\\
 &= I_{\OO_X}\!\left(
    \operatorname{ext}_0\bigl(
     \rho(t-R)\,d't\wedge d''t\wedge\eta|_{X\setminus D}\bigr)\right).
 \end{split}                                                       \tag{3.3}
\]
Every integrand in (3.3) is an ordinary smooth compactly supported top
form with support in \(\Int(X/K)\). The equalities are scalar equalities.
\end{proposition}
