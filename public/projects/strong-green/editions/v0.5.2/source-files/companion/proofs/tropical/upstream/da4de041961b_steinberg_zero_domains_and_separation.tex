\section{Native Steinberg loops: zero domains and strong passive separation}
\label{sec:r11-challenge-zero-domains}

\paragraph{Provenance and scope.}
This module is jointly constructed by the R11 root and challenge agents;
both independently obtained the strong-separation mechanism.  Neither
may be counted as its nonconstructor reviewer.  Its current status is
\textsc{proved draft, pending exact-hash independent review}.  The local
statements concern the literal frozen SHT pair, not an unspecified
primitive of its boundary.  They require no regular-sequence assumption.
The Green consequence below has separately stated geometric hypotheses.

Fix an ordinary original-$K$ analytic open $N$, a metrized line bundle,
and an $r$-tuple $f$ of its analytic sections, where $r\ge3$.  Let $i,j,k$
be distinct.  Put $W=N\setminus |f=0|$ and let $b,c$ be arbitrary analytic
functions on $N$, including functions with zeros and unbounded norm.
Put $s=r-1$ and reserve $D=d'd''$ for the full differential operator.
Write
\[
 B=E_{ij}(b),\qquad C=E_{jk}(c),\qquad E=E_{ik}(bc),
 \qquad BC=CEB.
 \tag{ZDS1}
\]
Matrices act on column tuples, with the rightmost matrix acting first.
Set
\[
 u=Cf,\qquad v=Bf,\qquad w=EBf,\qquad h=BCf=CEBf.
 \tag{ZDS2}
\]
If $\mathcal P_{ab,t}(g)$ denotes the frozen SHT native shear pair,
the prescribed defect is
\[
 \mathscr P=
 \mathcal P_{jk,c}(f)+\mathcal P_{ij,b}(u)
 -\mathcal P_{ij,b}(f)-\mathcal P_{ik,bc}(v)
 -\mathcal P_{jk,c}(w).
 \tag{ZDS3}
\]
For $t\ne0$, this native pair is the exact sum of two unit edges
with potentials
\[
 (\Phi_g,\Phi_{D_a(t)^{-1}g},-v(t)),\qquad
 (\Phi_{D_a(t)^{-1}g},\Phi_{E_{ab}(t)g},v(t)).
 \tag{ZDS4}
\]
The middle potential is equally the norm potential of
$D_a(t)^{-1}E_{ab}(t)g$, by the norm isometry $E_{ab}(1)$.
In particular, equal endpoint norm potentials make the \emph{entire}
native pair zero, for arbitrary unit coefficient $t$.  Frozen SHT
extends the pair by zero through $t=0$ using its literal vanishing on
$|t|<1$.

\subsection{The complete ten-edge presentation}

On $N_{bc}=\{bc\ne0\}$ put $d=v(b)$ and $e=v(c)$, and introduce
\[
 \begin{array}{llll}
 F=\Phi_f,& U=\Phi_{D_j(c)^{-1}f},& C_0=\Phi_u,
   & T=\Phi_{D_i(b)^{-1}u},\\
 H=\Phi_h,& J=\Phi_{D_j(c)^{-1}w},& W_0=\Phi_w,
   & K_0=\Phi_{D_i(bc)^{-1}v},\\
 V=\Phi_v,& S=\Phi_{D_i(b)^{-1}f}.&&
 \end{array}
 \tag{ZDS5}
\]
The complete oriented loop, including the direction of every reversed
edge, is
\[
\begin{array}{c|c|c}
\text{edge}&\text{ordered potentials}&\text{weight}\\\hline
1&(F,U)&-e\\
2&(U,C_0)&e\\
3&(C_0,T)&-d\\
4&(T,H)&d\\
5&(H,J)&-e\\
6&(J,W_0)&e\\
7&(W_0,K_0)&-(d+e)\\
8&(K_0,V)&d+e\\
9&(V,S)&-d\\
10&(S,F)&d.
\end{array}
\tag{ZDS6}
\]
This is a literal expansion of (ZDS3), not a replacement path.

\begin{lemma}[Coefficient-zero neighborhoods]
Define ordinary opens
\[
 O_b=\{|b|<1,\ |bc|<1\},\qquad
 O_c=\{|c|<1,\ |bc|<1\}.
 \tag{ZDS7}
\]
They contain the zero loci of $b,c$, respectively.  The actual pair
$\mathscr P$ is identically zero on $(O_b\cup O_c)\cap W$.
Moreover, on $N_{bc}$ each of these vanishing identities pairs all
ten edges in (ZDS6) into five exact inverse pairs.
\end{lemma}

\begin{proof}
On $O_b$, $B,E$ are norm isometries.  Consequently
\[
 F=V=W_0,\qquad C_0=H.
 \tag{ZDS8}
\]
On the $c$-unit locus,
\[
 D_j(c)^{-1}(EB)D_j(c)
       =E_{ik}(bc)E_{ij}(bc)
 \tag{ZDS9}
\]
is a product of norm isometries, so $U=J$.  Thus edges $(1,6)$,
$(2,5)$, $(3,4)$, $(7,8)$, $(9,10)$ are exact inverse pairs.
If $c=0$, its native shear pairs are zero on a neighborhood by SHT,
so the same actual identity holds without dividing by $c$ there.

On $O_c$, $C,E$ are norm isometries.  Therefore
\[
 F=C_0,\qquad V=W_0=H.
 \tag{ZDS10}
\]
Because $i,j,k$ are distinct, $D_i(b)^{-1}$ commutes with $C$.
Hence $S=T$ on the $b$-unit locus.  The inverse-edge pairs are now
$(1,2)$, $(3,10)$, $(4,9)$, $(5,6)$, $(7,8)$.
SHT zero gluing handles $b=0$.  Each assertion is equality of
norm potentials on an ordinary open, and hence equality of their
actual derivatives and all mixed jets, not merely pointwise
agreement of one chosen coefficient representative.
\end{proof}

\subsection{A strong passive-separation region}

Let the common passive tuple be
\[
 f_{\rm pas}=(f_a)_{a\notin\{i,j\}},
 \qquad M=\|f\|_\infty,\qquad P=\|f_{\rm pas}\|_\infty.
 \tag{ZDS11}
\]
It contains $f_k$ and has $r-2$ entries.  The metric is the same
in all these norms.  On $N_{bc}\cap W$ set
\[
 B_0=\max(0,-d),\qquad C_0^*=\max(0,-e,-d-e),\qquad
 \kappa=B_0+C_0^*.
 \tag{ZDS12}
\]
The notation $C_0^*$ denotes a scalar bound, not a curvature or the
potential $C_0$ in (ZDS5).  If desired it may be replaced by any
larger nonnegative actual piecewise affine bound.

\begin{lemma}[Strong passive separation gives literal loop cancellation]
On the ordinary open
\[
 \mathcal O_{\rm small}
 =\{P<e^{-\kappa}M\}
 =\{\psi>\Phi_f+\kappa\},\qquad \psi=-\log P,
 \tag{ZDS13}
\]
with $\psi=+\infty$ at $P=0$, one has
\[
 F=C_0,\qquad V=W_0=H,\qquad S=T.
 \tag{ZDS14}
\]
Thus the complete ten-edge loop cancels in the same five inverse
pairs as on $O_c$, even when $b,c,bc$ have arbitrarily large norm.
\end{lemma}

\begin{proof}
The inequalities are pointwise over the completed residue field of
every analytic point.  They are independent of any rational-point
density claim.  Put $L=\max(1,|b|)=e^{B_0}$ and
$A=\max(1,|c|,|bc|)=e^{C_0^*}$.  The hypothesis is
\[
                     AP<M/L.
 \tag{ZDS15}
\]
The inverse matrix $B^{-1}=E_{ij}(-b)$ has operator norm at most
$L$, so $\|v\|\ge M/L$.  Also
\[
 \|D_i(b)^{-1}f\|\ge M/L,
 \tag{ZDS16}
\]
because the inverse of this diagonal matrix has operator norm
$L$.  The differences of the relevant tuples satisfy
\[
 \begin{split}
 \|u-f\|&\le |c|P<M/L\le M,\\
 \|w-v\|&\le |bc|P<M/L\le\|v\|,\\
 \|h-v\|&\le\max(|c|,|bc|)P<M/L\le\|v\|.
 \end{split}
 \tag{ZDS17}
\]
The strong triangle law therefore gives $\|u\|=\|f\|$ and
$\|w\|=\|h\|=\|v\|$.

Finally $D_i(b)^{-1}$ leaves the $j$-th component unchanged and
commutes with $C$.  Thus the difference between
$D_i(b)^{-1}u$ and $D_i(b)^{-1}f$ is exactly $c f_k$ in its
$j$-th component.  Its norm is strictly less than $M/L$, while
(ZDS16) bounds the latter tuple from below.  Their norms agree.
This proves (ZDS14).  All inequalities define ordinary opens via
continuous norm ratios.  The equalities hold on the entire open,
so they retain the full fixed-presentation actual relations.
\end{proof}

 L_1^s=B^{s-1,s}\oplus B^{s,s-1},\qquad

\begin{theorem}[Native all-rank Steinberg relation]
For all $r\ge3$, original-$K$ data (ZDS1)--(ZDS3), and arbitrary
analytic $b,c$ on the stated whole ordinary open, there is an
explicit actual triple $q\in L_2^{r-1}(W)$ with
\[
 \boxed{\quad
 \mathcal P_{jk,c}(f)+\mathcal P_{ij,b}(Cf)
 -\mathcal P_{ij,b}(f)-\mathcal P_{ik,bc}(Bf)
 -\mathcal P_{jk,c}(EBf)=\partial_2q.
 \quad}
 \tag{ZDS38}
\]
On $\{bc\ne0\}\cap W$, $q$ is (ZDS18), (ZDS21),
(ZDS27)--(ZDS37).  On $(O_b\cup O_c)\cap W$, it is zero.
These formulas agree on the full ordinary overlaps and cover
$W$.  They include the original common metric, every absolute-value
and cap wall, zeros of individual tuple entries, zeros of $b,c$,
and all mixed coefficient jets.  No field extension, analytic
root, boundedness assumption, completion, choice of an abstract
primitive, or continuous linear right inverse is used.
\end{theorem}

\begin{proof}
The preceding lemmas prove the norm equalities and exact inverse-edge
cancellations.  Here is the remaining passive-block argument in the
precise generality needed, so that no square-specific hypothesis is
silently imported.  On $\mathcal O_{\rm pas}$ the reference curvature
$R_*=R-D\psi$ is the pullback of the standard form from the map
defined by the unchanged $r-2$ passive sections to $\mathbf P^{r-3}$.
Thus $R_*^{s-1}=R_*^{r-2}=0$.  For a unit edge of weight $c_e\ge0$,
frozen CS15 kills every $C_p$-containing monomial in its first
residual symbol.  The rest is
\[
          \mathfrak a(p+c_e,q;h_{s-1}(R_*,C_q)).
 \tag{ZDS39}
\]
The pure $R_*^{s-1}$ term is zero.  On a carrier of every other
monomial there is a positive power of $C_q$.  If the changed row
is active, $p+c_e=q$ with full tangential first-jet equality.  If
it is inactive and a passive row is active, $q=0$ with zero
tangential first jet.  If neither occurs, the other target row
is uniquely active, so $C_q$ vanishes on an ordinary neighborhood.
These alternatives exhaust all rows.  Reverse the edge for negative
$c_e$; on $c_e=0$ use $p=q$ and the zero restricted first jets of
$(c_e)_\pm$.  Hence $S_{e,s}=0$ for every one of the ten edges
on $\mathcal O_{\rm pas}$.  This is the full actual carrier
calculation of frozen CS, retaining arbitrary extra base directions.

The strong-separation lemma gives the full inverse pairing on
$\mathcal O_{\rm small}$, including the cap-transition part.
Consequently all terms in (ZDS20) and all $f_k,\Gamma_k$ vanish
there.  The two opens cover, so $\sum_eS_{e,s}=0$ globally on
the coefficient-unit locus.  Equations (ZDS31)--(ZDS37) prove
(ZDS38) on it.  The coefficient-zero lemma gives equality of
the \emph{specified triples} with zero on both overlaps, so
ordinary actual sheaf gluing proves the theorem on all $W$.
\end{proof}

\subsection{Two ordinary opens and the original-support Green form}

 G_{s_1}-G_{s_0}=\partial_1p_a+E_a\quad\text{on }N_a\cap V_0,

 g_{V_0}=G_{s_1}-G_{s_0}

\subsection{A rejected shortcut: mere inactivity of passive entries}

It is false that $\sum_eS_{e,1}=0$ follows just because every one
of the ten vertex norms is determined by entries $i,j$.  On a
local unit presentation put
\[
 x=v(f_i),\quad y=v(f_j),\quad v_1=v(f_j+cf_k),\quad
 w_1=v(f_i+bf_j+bcf_k).
 \tag{ZDS22}
\]
Using the ultrametric triangle relation on each native shear,
the degree-one residual on that open reduces, separately in each
colour, to
\[
 \begin{split}
 &\mathfrak s(F-C_0,x)+\mathfrak s(C_0-H,v_1)
   -\mathfrak s(F-W_0,y)-\mathfrak s(W_0-H,w_1)\\
 &\hspace{30mm}=\mathfrak s(x-w_1,v_1-y),
 \qquad\mathfrak s(a,b)=a\,db-b\,da.
 \end{split}
 \tag{ZDS23}
\]
For the equality use $F=\min(x,y)$, $C_0=\min(x,v_1)$,
$W_0=\min(w_1,y)$, $H=\min(w_1,v_1)$ and the full piecewise
linear identity
$\mathfrak s(\min(a,b),a-b)=-\mathfrak s(a,b)$.
The coefficient can be nonzero on a strict Gauss chamber: take
\[
 (v(f_i),v(f_j),v(f_k),v(b),v(c))=(-2,-3,0,-2,-5).
 \tag{ZDS24}
