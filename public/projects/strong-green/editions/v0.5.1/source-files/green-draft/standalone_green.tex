\title[Actual Green forms and Chow correspondences]{Actual Non-Archimedean Green Forms\\ for Abelian Cycles and Products of Curves}
\author{}
\date{16 November 2026}
\begin{document}
\begin{abstract}
Let $K/\mathbf Q_p$ be finite and let $X/K$ be smooth and
projective.  We construct actual higher Chern forms of vector bundles
on projective models, a finite trace on the module generated by
piecewise-linear scalar potentials and model curvature forms, and
Green representatives of supported perfect complexes.  For each
prescribed rational cycle $Z$ we obtain an actual Green form on
$X^{\an}\setminus |Z|^{\an}$ with its model curvature.  A compact
parameter argument then produces a global curvature primitive for
every geometrically algebraically trivial Chow class.  Consequently
such a prescribed cycle admits an ordinary actual zero-curvature
Green form, tested against every compact complementary actual
$\delta$-form over the original field.  Combining the numerical
curvature detector and finite trace with K\"unnemann's semistable
special-fiber theorem yields the same conclusion for every
geometrically homologically trivial cycle on an abelian variety.
A supported numerical projection formula then transports model
certificates through Chow correspondences.  This applies, in every
codimension, to products of arbitrary smooth projective curves and
abelian varieties and to varieties finitely dominated by such
products.  Native degrees, cycle coefficients, compact actual
tests, and the prescribed support are retained throughout.
\end{abstract}

\section{Introduction}\label{sec:introduction}

Fix a finite extension $K/\mathbf Q_p$ with its given absolute value,
and let $X/K$ be smooth, projective, and geometrically integral
of dimension $n$.  Write $D=d'd''$ and let
$\B^{p,q}$ be the sheaf of actual $\delta$-forms in the sense
of Mihatsch \cite{M}, on the ordinary topology of $X^{\an}$.
For a rational cycle $Z=\sum_j a_j[Z_j]$ of codimension $r$,
put $|Z|=\bigcup_{a_j\ne0} Z_j$.
Our zero-curvature Green identity is
\begin{equation}\label{eq:main-green}
\int_{X^{\an}\setminus|Z|^{\an}}\gamma_Z\wedge D\eta
=-\int_{Z^{\an}}\eta,
\qquad
\eta\in\B_c^{n-r,n-r}(X^{\an}).
\end{equation}
The integrals are defined in the actual calculus, and the right
side includes the rational coefficients and generic lengths of $Z$.

\begin{theorem}[Algebraic equivalence]\label{thm:main-algebraic}
Let $1\le r\le n$ and let $Z$ be a prescribed rational
codimension-$r$ cycle on $X$.
If its Chow class is algebraically equivalent to zero over
$\overline K$, there exists
\[
\gamma_Z\in\B^{r-1,r-1}
       (X^{\an}\setminus |Z|^{\an})
\]
satisfying \eqref{eq:main-green} for every compact actual
complementary test, over $K$.
\end{theorem}

The proof has two geometric steps and one analytic step.
A supported perfect complex representing the given cycle becomes,
after a generically finite map which is finite flat on the prescribed
complement, a finite sum of two-term complexes of line bundles.
Their scalar Poincar\'e--Lelong identities yield a Green pair.
Actual finite trace carries this pair back to the prescribed open.
The curvature class vanishes on algebraically trivial cycles by
rigidity in a proper $p$-adic parameter family and compactness of
the rational points of an abelian variety.

The analytic construction also gives unconditional results in two
reduction situations.
\begin{theorem}[Model-curvature Green pairs and good reduction]
\label{thm:main-pairs}
For each prescribed rational codimension-$r$ cycle $Z$ on $X$
there exist a separately closed actual model Chern form
$\Omega_Z\in\B^{r,r}(X^{\an})$ and an actual Green form
$G_Z\in\B^{r-1,r-1}(X^{\an}\setminus|Z|^{\an})$ such that
\begin{equation}\label{eq:main-pair}
\int_{X^{\an}\setminus|Z|^{\an}}G_Z\wedge D\eta+
\int_{Z^{\an}}\eta
=\int_{X^{\an}}\Omega_Z\wedge\eta
\end{equation}
for all $\eta\in\B_c^{n-r,n-r}(X^{\an})$.
If $X$ has smooth projective good reduction, the right-hand side
vanishes for geometrically homologically trivial $Z$.
The same conclusion holds if $X$ acquires such reduction after
a finite extension of $K$.
\end{theorem}

The assertion about good reduction follows from numerical vanishing
of model Chern forms.  The finite trace is constructed on the
scalar--curvature module generated by PL potentials and Chern forms.

\begin{theorem}[Abelian cycles and finite abelian domination]
\label{thm:main-abelian}
Let $A/K$ be an abelian variety.  For every prescribed rational
cycle $Z$ whose class is geometrically homologically trivial,
the zero-curvature Green identity \eqref{eq:main-green} has an
ordinary actual solution on its exact support complement.
More generally, the same assertion holds for a smooth projective
geometrically integral $X/K$ admitting a finite surjective
morphism from an abelian variety over $K$, in every codimension.
\end{theorem}

We first construct a numerical certificate on a semistable model
using K\"unnemann's special-fiber exact sequence.  The numerical
detector makes the associated actual curvature vanish strictly.
Finite trace then transfers a structured primitive to the original
field, where the exact-support Green-pair construction completes
the proof.  All of these steps are proved in this paper.

Numerical classes on regular models admit a proper projection
formula.  With regular alterations, it gives a transfer theorem
for Chow correspondences.  The actual curvature detector reads
the resulting numerical certificates.

\begin{theorem}[Products of curves and abelian varieties]
\label{thm:main-curve-products}
Let $X=A\times_K C_1\times_K\cdots\times_K C_s$, where $A/K$
is an abelian variety and every $C_i/K$ is a smooth projective
geometrically integral curve.  In every codimension, each
prescribed rational cycle $Z$ with vanishing geometric $\ell$-adic
cycle class has an actual zero-curvature Green form on
$X^{\an}\setminus|Z|^{\an}$ satisfying \eqref{eq:main-green}
over the original $K$.  The same conclusion holds for smooth
projective geometrically integral varieties finitely dominated
by such a product.
\end{theorem}

The proof uses a Chow-diagonal decomposition through a product
of Jacobians, a numerical correction on an abelian model,
and Hartl's construction of regular models for products.

\section{Foundational conventions and analytic inputs}
\label{sec:foundations}

\paragraph{Models, forms, and current pairings.}
For a model $\mathcal X/K^\circ$ a model line bundle
$\mathcal L$ determines a PL metric on its generic fiber $L$.
Its actual curvature $c_1(\overline L)=D(-\log\|e\|)$
belongs to the pure tridegree $\B^{0,0,1}$.
A pure form of tridegree $(0,0,r)$ has bidegree $(r,r)$.
All special-fiber intersections use rational operational Chow classes.
Native tropical calibrations and multiplicities are those of
\cite{M}; they are retained in every integration and degree.
We abbreviate $\B_c^{p,q}(X^{\an})$ for the compactly supported
actual tests.

\paragraph{Published analytic results.}
We use Mihatsch's ordinary sheaf and analytic pullback theorem
\cite[Definition~4.2 and Theorem~4.5]{M}, the witnessing-skeleton
criterion, restriction and support statements
\cite[Lemma~4.4, Corollary~4.9 and Proposition~4.11]{M},
his native finite-flat top-degree identity
\cite[Lemma~3.15 and Definition~4.10]{M}, and scalar strong
Poincar\'e--Lelong \cite[Theorem~5.5 and Corollary~5.6]{M}.
For polyhedral projection and pullback we use
\cite[Section~2 and Proposition/Definition~4.2]{MTrop} and
\cite[Proposition~2.38 and Definition~2.41]{M}.
These foundational statements supply the actual differential and
integral calculus in which the proofs take place.

For the scalar density argument we use the uniform density of
rational model functions on a proper analytification
\cite[Proposition~2.15]{GM}.  For intersections of model metrics
we use the recursive definition and native component-mass formula
of Chambert-Loir--Ducros
\cite[Sections~8.7--8.8 and Theorem~18.5.5]{CLD}.
Our use of localized Riemann--Roch is the Cartier case of
\cite[Corollary~18.1.2]{Fulton}.
All varieties to which the actual sheaf is applied are
pure-dimensional and boundaryless.  All finite maps and
projective models are over the original field $K$ unless an
extension is displayed explicitly.

\paragraph{Algebraic inputs.}
We use the rational Chern character description of $K_0$ on
regular schemes, resolution by vector bundles on smooth projective
varieties, proper and refined Gysin formulas for supported
perfect complexes \cite{Stacks}, Bertini and miracle flatness
\cite{Stacks}, and resolution of singularities for varieties
over characteristic-zero fields.  Algebraic equivalence is
generated by degree-zero divisor correspondences from smooth
projective curves \cite{ACMV}.  For a pointed curve, the
normalized Poincar\'e bundle on the curve times its Jacobian
provides an additive family \cite{MilneJV}.

\section{Proofs of the introductory theorems}

\begin{proof}[Proof of Theorem~\ref{thm:main-pairs}]
The exact-support curved Green identity is
Corollary~\ref{r11p:any-cycle-pair} together with
Theorem~\ref{r11p:complex-green}.
On a smooth projective good model the curvature vanishes by
Corollary~\ref{r11p:good-reduction-G}.
For potential good reduction apply
Corollary~\ref{r11p:potential-good-G}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main-algebraic}]
Theorem~\ref{r12p:geometric-algebraic} provides a global actual
form $\Psi_Z$ satisfying $D\Psi_Z=\Omega_Z$ for the
curvature of Corollary~\ref{r11p:any-cycle-pair}.
Set $\gamma_Z=G_Z-\Psi_Z|_{X^{\an}\setminus|Z|^{\an}}$.
The identity \eqref{eq:main-pair} and actual Stokes yield
\eqref{eq:main-green} on every stated test.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main-abelian}]
The first assertion is \Cref{thm:abelian-G}; the second is
Corollary~\ref{cor:finite-abelian}.  Both proofs use the finite numerical
descent theorem \Cref{thm:finite-numerical}, whose analytic inputs
were established in the preceding sections.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main-curve-products}]
Apply Theorem~\ref{thm:curve-products-G} and
Corollary~\ref{cor:finite-curve-product}.
The supported projection and correspondence theorems of
\Cref{sec:correspondences} supply the algebraic certificates.
\end{proof}
